# 八大员
```javascript
(function(){ let result = []; let total = parseInt(document.getElementById('quesOrderDiv').innerText.match(/共 (\d+) 题/)[1]); function getOneQues() { let trunk = document.querySelector('.trunk')?.innerText || ''; let opts = Array.from(document.querySelectorAll('.opt label')).map(l => l.innerText); let ansSpan = document.querySelector('span[a]'); let ansVal = ansSpan?.getAttribute('a'); let ansLabel = ''; document.querySelectorAll('.opt input').forEach(i => { if(i.value == ansVal) ansLabel = i.getAttribute('optno'); }); return { 题目: trunk, 选项: opts.join('\n'), 答案: ansLabel }; } function delayNext() { return new Promise(r => setTimeout(r, 800)); } async function run() { for(let i=0; i<total; i++){ let q = getOneQues(); result.push(q); console.log(`已抓取：${i+1}/${total}`); toNextQues(); await delayNext(); } let text = result.map((q,i) => `【第${i+1}题】 ${q.题目} ${q.选项} 答案：${q.答案} ----------------------------------------` ).join('\n'); let blob = new Blob([text], {type:'text/plain'}); let a = document.createElement('a'); a.href = URL.createObjectURL(blob); a.download = '土建请您质量员题库.txt'; a.click(); } run(); })();
```
